CREATE VIEW employee_details AS
SELECT
    e.employee_id,
    e.employee_name,
    e.email,
    e.salary,
    d.department_name
FROM employees e
JOIN departments d
	ON e.department_id = d.department_id;

CREATE VIEW it_employees AS
SELECT
    e.employee_id,
    e.employee_name,
    e.email,
    e.salary,
    d.department_name
FROM employees e
JOIN departments d
	 ON e.department_id = d.department_id
WHERE d.department_name = 'IT';

SELECT * FROM it_employees;
